select w.wonum, w.location, w.plusgsafecrielem, w.targcompdate, w.estdur
from workorder w
where w.woclass = 'BYPASS' and w.status = 'INPRG' and w.plusgsafecrielem is not null
order by w.targcompdate;

select c.consqnum, c.severity, r.likelihood, r.riskrank, r.riskpriority
from plusgconsq c join plusgrisklist r on r.plusgconsqid = c.plusgconsqid
where c.status = 'ACTIVE'
order by c.displaysequence, r.rowsequence;
