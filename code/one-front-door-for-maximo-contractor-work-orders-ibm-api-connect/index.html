# CONTRACTOR_COMPLETE: Maximo REST endpoint POST /maximo/api/script/contractor_complete
# Called only by API Connect. Body: {"wonum": "...", "completedAt": "...", "note": "..."}
# Header x-contractor-vendor: set by the gateway from the caller's consumer organization.
# Implicit variables (script with no launch point, called over REST): request, requestBody, responseBody, userInfo
from psdi.server import MXServer
from psdi.mbo import SqlFormat
from psdi.util import MXException
from com.ibm.json.java import JSONObject
from java.util.regex import Pattern  # Maximo's Jython has no Python standard library

def obj(d):
    o = JSONObject()
    for k, v in d.items():
        o.put(k, obj(v) if isinstance(v, dict) else v)
    return o

def reply(d):
    global responseBody
    responseBody = obj(d).serialize()

def error(code, message):
    reply({"Error": {"reasonCode": code, "message": message}})

vendor = request.getHeader("x-contractor-vendor") or ""
body = JSONObject.parse(requestBody or "{}")
wonum = body.get("wonum") or ""
note = (body.get("note") or "")[:2000]
completedAt = (body.get("completedAt") or "")[:30]

mx = MXServer.getMXServer()
woSet = mx.getMboSet("WORKORDER", userInfo)
try:
    if not woSet.getProfile().getAppOptionAuth("CONTRWO", "COMPLETE", None):
        error("FORBIDDEN", "User " + userInfo.getUserName() + " lacks CONTRWO COMPLETE")
    elif not Pattern.matches("[A-Z0-9]{1,12}", vendor) or not Pattern.matches("[A-Z0-9-]{1,25}", wonum):
        error("BADINPUT", "Missing or invalid vendor or work order number")
    else:
        # Bind variables, never string concatenation: the vendor check is part of the query
        sf = SqlFormat(userInfo, "wonum = :1 and vendor = :2 and istask = 0")
        sf.setObject(1, "WORKORDER", "WONUM", wonum)
        sf.setObject(2, "WORKORDER", "VENDOR", vendor)
        woSet.setWhere(sf.format())
        wo = woSet.moveFirst()
        if wo is None:
            error("NOTFOUND", "Not found")
        else:
            try:
                # The status memo holds 50 characters, so the contractor's note goes into a work log entry
                wo.changeStatus("COMP", mx.getDate(), ("Completed by contractor " + vendor)[:50])
                if note or completedAt:
                    log = wo.getMboSet("WORKLOG").add()
                    log.setValue("LOGTYPE", "CLIENTNOTE")
                    log.setValue("DESCRIPTION", ("Contractor " + vendor + " completion report")[:100])
                    log.setValue("DESCRIPTION_LONGDESCRIPTION", "Completed at (contractor time): " + (completedAt or "not given") + "\n" + note)
                woSet.save()
                reply({"number": wonum, "state": "done", "maximoStatus": wo.getString("STATUS")})
            except MXException, e:
                error("NOTALLOWED", e.getMessage())
finally:
    woSet.close()

<!--
  Contractor API: the Maximo side of an API Connect front door.

    CONTRWO    object structure: WORKORDER only, 13 fields, secured by its own application CONTRWO
    CONTRWO    sigoptions READ (read work orders) and COMPLETE (report completion through the
               CONTRACTOR_COMPLETE script, which checks this option itself)
    CONTRAPI   security group for the integration user API Connect signs in as: CONTRWO READ + COMPLETE,
               all sites, nothing else
    WORKORDER_NDX_VENDOR  index on WORKORDER (VENDOR, SITEID, STATUS): every gateway query filters on vendor

  Field list in MAXINTOBJCOLS: with EXCLUDEBYDEFAULT = 1 on the WORKORDER detail row, the rows listed as
  EXCLUDE are the fields that are INCLUDED (same encoding as IBM's MXAPIWOCHANGESTATUS).

  Install: copy to tools/maximo/en/script in the maxinst pod, then from tools/maximo/internal
           ./runscriptfile.sh -cscript -fcontractor_api_1.0.0
  Re-runnable: every insert uses ignore_duplicates.
  After install: reload the INTOBJECT cache (or restart the Manage server bundle) so the new structure is served.

"""Demo data for the contractor API: three contractor companies and the BEDFORD work orders assigned to them.

Re-runnable: companies are created only if missing; work orders are created only if no open work order with
the same description and vendor exists. Uses the Manage REST API (MXAPIVENDORMSTR, MXAPIVENDOR, MXAPIWODETAIL).

    set MAXIMO_API=https://<manage-host>/maximo/api
    set MAXIMO_APIKEY=<an administrator's API key, for the demo data only>
    python seed_demo.py
"""
import datetime as dt
import json
import os
import urllib.error
import urllib.parse
import urllib.request

class m:  # minimal Maximo REST client
    BASE, KEY = os.environ["MAXIMO_API"].rstrip("/"), os.environ["MAXIMO_APIKEY"]

    @staticmethod
    def call(method, path, body=None, headers=None):
        h = {"apikey": m.KEY, "Accept": "application/json", "Content-Type": "application/json"}
        h.update(headers or {})
        r = urllib.request.Request(m.BASE + path, method=method, headers=h,
                                   data=json.dumps(body).encode() if body is not None else None)
        try:
            with urllib.request.urlopen(r, timeout=120) as f:
                t = f.read()
                return f.status, (json.loads(t) if t else {})
        except urllib.error.HTTPError as e:
            return e.code, json.loads(e.read() or b"{}")

    @staticmethod
    def find(os_name, where, select="*"):
        q = urllib.parse.urlencode({"lean": 1, "oslc.select": select, "oslc.where": where, "oslc.pageSize": 100})
        code, r = m.call("GET", f"/os/{os_name}?{q}")
        if code != 200:  # never treat a failed query as "not found"
            raise SystemExit(f"query on {os_name} failed: {code} {json.dumps(r)[:300]}")
        return r.get("member", [])

def ensure_vendor(code, name):
    found = m.find("mxapivendor", f'company="{code}" and orgid="{ORG}"', "company")
    if found:
        return "exists"
    if not m.find("mxapivendormstr", f'company="{code}"', "company"):
        c0, r0 = m.call("POST", "/os/mxapivendormstr?lean=1", {"company": code, "name": name, "type": "V", "companysetid": "COMPSET1"})
        if c0 not in (200, 201):
            return f"master FAILED {c0} {json.dumps(r0)[:300]}"
    code_, r = m.call("POST", "/os/mxapivendor?lean=1", {
        "company": code, "name": name, "type": "V", "orgid": ORG, "companysetid": "COMPSET1", "currencycode": "USD"})
