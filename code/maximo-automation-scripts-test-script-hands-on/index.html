# BLOGTEST_WOSUMMARY - REST API script (no launch point), read-only.
# Call: GET /maximo/api/script/BLOGTEST_WOSUMMARY?site=BEDFORD&status=APPR
from psdi.server import MXServer
from psdi.mbo import SqlFormat
from com.ibm.json.java import JSONObject, JSONArray
site = request.getQueryParam("site") or "BEDFORD"
status = request.getQueryParam("status") or "APPR"
service.log("Request received: site=" + site + ", status=" + status)
woSet = MXServer.getMXServer().getMboSet("WORKORDER", request.getUserInfo())
try:
    sql = SqlFormat(request.getUserInfo(), "siteid = :1 and status = :2 and istask = 0")
    sql.setObject(1, "WORKORDER", "SITEID", site)
    sql.setObject(2, "WORKORDER", "STATUS", status)
    woSet.setWhere(sql.format())
    woSet.setOrderBy("wonum")
    total = woSet.count()
    service.log("Matching work orders: " + str(total))
    sample = JSONArray()
    wo = woSet.moveFirst()
    while wo is not None and sample.size() < 3:
        row = JSONObject()
        row.put("wonum", wo.getString("WONUM"))
        row.put("description", wo.getString("DESCRIPTION"))
        row.put("priority", wo.getString("WOPRIORITY"))
        sample.add(row)
        wo = woSet.moveNext()
finally:
    woSet.close()
result = JSONObject()
result.put("site", site)
result.put("status", status)
result.put("count", total)
result.put("firstThree", sample)
responseBody = result.serialize(True)

# BLOGTEST_WOPRIORITY - attribute launch point, WORKORDER.WOPRIORITY, Validate.
# Rule: emergency work (work type EM) must have priority 1 or 2.
# During Validate, mbo already holds the new value the user typed.
wonum = mbo.getString("WONUM")
worktype = mbo.getString("WORKTYPE")
priority = mbo.getString("WOPRIORITY")
service.log("Validating WO " + wonum + ": work type=" + worktype + ", new priority=" + priority)
if worktype == "EM" and (mbo.isNull("WOPRIORITY") or mbo.getInt("WOPRIORITY") > 2):
    service.log("Rule broken: EM work needs priority 1 or 2")
    service.error("configure", "BlankMsg", ["Emergency (EM) work orders need priority 1 or 2."])
service.log("Rule passed")

# BLOGTEST_WOINIT - object launch point, WORKORDER, Initialize Value.
# Gives every NEW work order a default priority of 3.
# The Initialize event also fires when an existing record is loaded,
# so the script only acts on records that are being added.
if mbo.toBeAdded():
    mbo.setValue("WOPRIORITY", 3)
    service.log("New work order at site " + mbo.getString("SITEID") + ": default priority 3 set")
else:
    service.log("Existing work order " + mbo.getString("WONUM") + ": nothing to do")
