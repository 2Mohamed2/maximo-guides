# RBI calculation core, structured after API RP 581 (3rd edition) Part 1 and Part 2.
# Pure functions only: this text is embedded in the RBI.CALC automation script (Jython 2.7)
# and imported by the test harness (CPython 3), so keep it compatible with both.
import math

# ---- Standard constants -------------------------------------------------------------------------------
# Values follow API RP 581 3rd edition (Part 2, thinning model; Part 1, risk categories).
# VERIFY them against your licensed copy of API RP 581 before production use and keep them under change control.
DS = (1.0, 2.0, 4.0)                       # damage state factors: rate as measured, 2x, 4x
COV_DT, COV_SF, COV_P = 0.20, 0.20, 0.05   # coefficients of variation: thinning, flow stress, pressure
PRIOR = {"LOW": (0.5, 0.3, 0.2), "MEDIUM": (0.7, 0.2, 0.1), "HIGH": (0.8, 0.15, 0.05)}
COND = {"A": (0.9, 0.09, 0.01), "B": (0.7, 0.2, 0.1), "C": (0.5, 0.3, 0.2), "D": (0.4, 0.33, 0.27)}
NORM = 1.56e-4                             # thinning damage factor normalisation
POF_DF = (1.0, 10.0, 100.0, 1000.0)        # PoF category 1..5 upper limits on the damage factor
COF_FIN = (1e4, 1e5, 1e6, 1e7)             # CoF category A..E upper limits, financial (USD)
COF_AREA = (9.29, 92.9, 929.0, 9290.0)     # CoF category A..E upper limits, consequence area (m2) = 100..100000 ft2

# ---- Screening values (company-configurable, not API tables) ------------------------------------------
# API RP 581 gives mechanism-specific base damage factor tables for cracking, HTHA and fatigue.
# These screening values stand in for them until you load the licensed tables.
SVI = {"HIGH": 50.0, "MEDIUM": 10.0, "LOW": 1.0}           # SCC severity index by susceptibility
BASE_OTHER = {"HIGH": 500.0, "MEDIUM": 50.0, "LOW": 5.0}   # HTHA / mechanical fatigue base damage factor
CREDIT = {"A": 0.1, "B": 0.25, "C": 0.5, "D": 0.8}          # factor per past inspection, by effectiveness
MAX_INTERVAL = 10.0                                          # years, absolute cap on an inspection interval
RISK_LEVEL = ((4, "LOW"), (6, "MEDIUM"), (7, "MEDHIGH"), (10, "HIGH"))  # matrix score = PoF cat + CoF index

def _erf(x):
    # Abramowitz-Stegun 7.1.26, |error| < 1.5e-7; used when math.erf is missing (older Jython)
    t = 1.0 / (1.0 + 0.3275911 * abs(x))
    y = 1.0 - (((((1.061405429 * t - 1.453152027) * t) + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t * math.exp(-x * x)
    return y if x >= 0 else -y

def phi(x):
    """Standard normal cumulative distribution."""
    erf = getattr(math, "erf", _erf)
    return 0.5 * (1.0 + erf(x / math.sqrt(2.0)))

def posterior(conf, counts):
    """Bayesian update of the damage-state priors with the inspections done (counts = dict A..D -> n)."""
    pr = PRIOR.get(conf, PRIOR["LOW"])
    w = []
    for i in range(3):
        v = pr[i]
        for k in ("A", "B", "C", "D"):
            v *= COND[k][i] ** counts.get(k, 0)
        w.append(v)
    s = sum(w)
    return [x / s for x in w]

def strength_ratio(t_rdi, tmin, s_allow, joint, ys, ts):
    """SR = (S*E / FS) * tmin / t_rdi with flow stress FS = (YS+TS)/2 * E * 1.1."""
    fs = (ys + ts) / 2.0 * joint * 1.1
    return (s_allow * joint / fs) * (tmin / t_rdi)

def df_thinning(cr, age, t_rdi, sr, conf, counts):
    """Thinning damage factor: probability of failure over the three damage states, normalised."""
    if cr <= 0 or t_rdi <= 0:
        return 0.0, 0.0
    art = cr * age / t_rdi
    post = posterior(conf, counts)
    p = 0.0
    for i in range(3):
        d = DS[i]
        num = 1.0 - d * art - sr
        den = math.sqrt((d * art * COV_DT) ** 2 + ((1.0 - d * art) * COV_SF) ** 2 + (sr * COV_P) ** 2)
        p += post[i] * phi(-num / den)
    return max(p / NORM, 0.1), art

def df_scc(susc, age, counts):
    """Cracking: base from the severity index, reduced by inspections, grown with age^1.1."""
    base = SVI.get(susc, 0.0)
    for k, n in counts.items():
        base *= CREDIT.get(k, 1.0) ** n
    return base * max(age, 1.0) ** 1.1

def df_other(susc, counts):
    base = BASE_OTHER.get(susc, 0.0)
    for k, n in counts.items():
        base *= CREDIT.get(k, 1.0) ** n
    return base

def pof_category(df):
    for i, lim in enumerate(POF_DF):
        if df <= lim:
            return i + 1
    return 5

def cof_category(value, limits):
    for i, lim in enumerate(limits):
        if value <= lim:
            return "ABCDE"[i]
    return "E"

def risk_level(pofcat, cofcat):
    score = pofcat + "ABCDE".index(cofcat) + 1
    for lim, lvl in RISK_LEVEL:
        if score <= lim:
            return lvl
    return "HIGH"

def fms(score):
    """Management systems factor from the management systems evaluation score (0-1000)."""
    return 10.0 ** (-0.02 * (score / 10.0) + 1.0)

def total_df(eq, mechs, age_rd, extra=None):
    """Damage factors at a time point.
    eq: dict(cr, t_rdi, sr, conf, counts) for internal thinning; age_rd = years since the thickness reading.
    mechs: list of dict(mech, susc, cr, age, counts); their 'age' is already measured to the same time point.
    extra: (mechanism, effectiveness) -> one more inspection credited to that mechanism (mitigation case).
    Combination rule (general thinning): max(internal, external/CUI) + cracking + HTHA + fatigue."""
    def cnt(c, mech):
        c = dict(c)
        if extra and extra[0] == mech:
            c[extra[1]] = c.get(extra[1], 0) + 1
        return c
    d_thin, art = df_thinning(eq["cr"], age_rd, eq["t_rdi"], eq["sr"], eq["conf"], cnt(eq["counts"], "THINNING"))
    rows, d_ext, d_scc, d_oth = [], 0.0, 0.0, 0.0
    for m in mechs:
        c = cnt(m["counts"], m["mech"])
        if m["mech"] in ("EXTCORR", "CUI"):
            d = df_thinning(m["cr"], m["age"], eq["t_rdi"], eq["sr"], "LOW", c)[0]
            d_ext = max(d_ext, d)
        elif m["mech"] == "SCC":
            d = df_scc(m["susc"], m["age"], c)
            d_scc = max(d_scc, d)
        else:
            d = df_other(m["susc"], c)
            d_oth += d
        rows.append(d)
    total = max(d_thin, d_ext) + d_scc + d_oth
    return dict(thin=d_thin, art=art, ext=d_ext, scc=d_scc, other=d_oth, total=total, rows=rows)

def plan(eq, mechs, age_rd0, gff, fmsv, fc, target, plan_years, rl, since_insp):
    """Risk-target inspection planning on a monthly grid from the RBI date (t=0) to the plan date and beyond.
    mechs carry age0 (their age at the RBI date). rl = remaining life at the RBI date (years);
    since_insp = years from the last inspection to the RBI date.
    Returns the inspection due time (years from the RBI date), what governs it, and the damage factors."""
    def at(t, extra=None):
        ms = [dict(m, age=m["age0"] + t) for m in mechs]
        return total_df(eq, ms, age_rd0 + t, extra)
    def risk(t, extra=None):
        return gff * fmsv * at(t, extra)["total"] * fc
    t_target, t = None, 0.0
    while t <= 15.0:
        if risk(t) > target:
            t_target = t
            break
        t += 1.0 / 12.0
    cands = [(MAX_INTERVAL - since_insp, "maximum interval of %.0f y" % MAX_INTERVAL),
             (rl / 2.0, "half the remaining life (API 510/570)")]
    if t_target is not None:
        cands.append((t_target, "risk target reached"))
    due, why = sorted(cands, key=lambda c: c[0])[0]
    due = max(due, 0.0)
    now, end = at(0.0), at(plan_years)
    drivers = [("THINNING", end["thin"])] + [(m["mech"], d) for m, d in zip(mechs, end["rows"])]
    driver = sorted(drivers, key=lambda x: -x[1])[0][0]
    need, mit = "A", None
    for eff in ("D", "C", "B", "A"):
        r = risk(plan_years, (driver, eff))
        if r <= target:
            need, mit = eff, r
            break
    if mit is None:
        mit = risk(plan_years, (driver, "A"))
    return dict(now=now, end=end, risk_now=risk(0.0), risk_plan=risk(plan_years), t_target=t_target, due=due,
                why=why, driver=driver, need=need, risk_mit=mit)

# ---- Maximo glue: read the assessment, run the model above, write the results ----------------------------
# Runs before every save of RBIASSESS (object launch point) and from Select Action > Recalculate Risk.
from psdi.mbo import MboConstants
from java.util import Date, Calendar

F = MboConstants.NOACCESSCHECK | MboConstants.NOVALIDATION_AND_NOACTION
YEAR_MS = 365.25 * 24 * 3600 * 1000.0
METHOD = {"THINNING": "UT thickness survey at all TMLs plus scanning of low points",
          "EXTCORR": "External visual inspection with spot UT at supports",
          "CUI": "Strip damaged insulation; profile RT or pulsed eddy current",
          "SCC": "Wet fluorescent MT or TOFD at welds and nozzles",
          "HTHA": "Phased-array UT and TOFD of welds for HTHA",
          "FATIGUE": "MT/PT at small-bore connections plus vibration survey"}

def years(d1, d2):
    return (d2.getTime() - d1.getTime()) / YEAR_MS

def add_years(d, y):
    cal = Calendar.getInstance()
    cal.setTime(d)
    cal.add(Calendar.DAY_OF_YEAR, int(round(y * 365.25)))
    return cal.getTime()

def setv(o, attr, val):
    if val is None:
        o.setValueNull(attr, F)
    else:
        o.setValue(attr, val, F)

def counts(o, attrs):
    return dict((k, o.getInt(a)) for k, a in zip(("A", "B", "C", "D"), attrs))

# Approved and superseded assessments are frozen records (API RP 580: the approved basis stays as approved).
# The calculation runs on drafts, and once more on the save that approves the assessment.
_st = mbo.getString("STATUS")
_frozen = _st == "SUPERSEDED" or (_st == "APPR" and not mbo.isModified("STATUS"))
if _frozen and launchPoint == "RBICALC":
    service.error("rbi", "frozen")

def _run():
    a = mbo
    if a.isNull("ASSESSDATE"):
        setv(a, "ASSESSDATE", Date())
    rbi = a.getDate("ASSESSDATE")
    if a.isNull("PLANDATE"):
        setv(a, "PLANDATE", add_years(rbi, 5.0))
    plan_y = max(years(rbi, a.getDate("PLANDATE")), 0.0)
    tnom = a.getDouble("TNOM")
    tmin = a.getDouble("TMIN")
    inst = a.getDate("INSTALLDATE")
    notes = []

    # 1. thickness history up to the RBI date, oldest first (gauge meter readings are MEASUREMENT rows)
    pts = []
    rs = a.getMboSet("$rbithk", "MEASUREMENT", "assetnum=:assetnum and siteid=:siteid and metername=:tmeter")
    rs.setOrderBy("measuredate")
    rs.reset()
    r = rs.moveFirst()
    while r is not None:
        d = r.getDate("MEASUREDATE")
        if d is not None and not r.isNull("MEASUREMENTVALUE") and not d.after(rbi):
            pts.append((d, r.getDouble("MEASUREMENTVALUE")))
        r = rs.moveNext()
    setv(a, "NUMREADINGS", len(pts))

    # 2. corrosion rates: long-term from nominal, short-term from the last two readings; the higher one governs
    crlt, crst = 0.0, 0.0
    if pts:
        dlast, tlast = pts[-1]
        if inst is not None and years(inst, dlast) > 0 and tnom > 0:
            crlt = (tnom - tlast) / years(inst, dlast)
        if len(pts) > 1 and years(pts[-2][0], dlast) > 0:
            crst = (pts[-2][1] - tlast) / years(pts[-2][0], dlast)
        setv(a, "TLAST", tlast)
        setv(a, "LASTREADDATE", dlast)
        if a.isNull("LASTINSPDATE"):
            setv(a, "LASTINSPDATE", dlast)
    else:
        dlast, tlast = (inst or rbi), tnom
        setv(a, "TLAST", None)
        setv(a, "LASTREADDATE", None)
        notes.append("No thickness readings: nominal thickness used.")
    cr = max(crlt, crst, 0.0)
    age_rd0 = max(years(dlast, rbi), 0.0)
    ttoday = tlast - cr * age_rd0
    rl = 99.0 if cr <= 0 else max(0.0, min(99.0, (ttoday - tmin) / cr))
    for k, v in (("CRLT", round(crlt, 3)), ("CRST", round(crst, 3)), ("CORRRATE", round(cr, 3)),
                 ("TTODAY", round(ttoday, 2)), ("REMLIFE", round(rl, 1))):
        setv(a, k, v)

    # 3. strength ratio from the design data (conservative S*E/FS = 1 when design data is missing)
    s_allow, ys, ts, joint = a.getDouble("ALLOWSTRESS"), a.getDouble("YIELDSTR"), a.getDouble("TENSILESTR"), a.getDouble("JOINTEFF") or 1.0
    if s_allow > 0 and ys > 0 and ts > 0 and tlast > 0:
        sr = strength_ratio(tlast, tmin, s_allow, joint, ys, ts)
    else:
        sr = tmin / tlast if tlast > 0 else 1.0
        notes.append("Design stresses missing: SR = tmin / t_rdi.")
    setv(a, "SR", round(sr, 3))

    # 4. damage mechanisms
    eq = dict(cr=cr, t_rdi=tlast, sr=sr, conf=a.getString("CRCONF") or "LOW",
              counts=counts(a, ("THINNA", "THINNB", "THINNC", "THINND")))
    mechs, rows = [], []
    dm = a.getMboSet("DAMAGE")
    d = dm.moveFirst()
    while d is not None:
        if not d.toBeDeleted():
            start = d.getDate("STARTDATE") or inst or rbi
            mechs.append(dict(mech=d.getString("MECHANISM"), susc=d.getString("SUSCEPT"), cr=d.getDouble("CORRRATE"),
                              age0=max(years(start, rbi), 0.0), counts=counts(d, ("NA", "NB", "NC", "ND"))))
            rows.append(d)
        d = dm.moveNext()

    # 5. probability, consequence, risk and the inspection plan
    gff = a.getDouble("GFF") or 3.06e-5
    fmsv = fms(a.getInt("MSSCORE"))
    fc = (a.getDouble("REPAIRCOST") + a.getDouble("AFFACOST") + a.getDouble("OUTAGEDAYS") * a.getDouble("LOSSPERDAY")
          + a.getDouble("INJCOST") + a.getDouble("ENVCOST"))
    target = a.getDouble("RISKTARGET") or 10000.0
    last_insp = a.getDate("LASTINSPDATE") or dlast
    p = plan(eq, mechs, age_rd0, gff, fmsv, fc, target, plan_y, rl, max(years(last_insp, rbi), 0.0))
    now, end = p["now"], p["end"]
    for row, dnow, dend in zip(rows, now["rows"], end["rows"]):
        setv(row, "DF", round(dnow, 1))
        setv(row, "DFPLAN", round(dend, 1))
        setv(row, "RECMETHOD", METHOD.get(row.getString("MECHANISM"), ""))
    pc_now, pc_end = pof_category(now["total"]), pof_category(end["total"])
    cfin, carea = cof_category(fc, COF_FIN), cof_category(a.getDouble("CONSAREA"), COF_AREA)
    cof = max(cfin, carea)
    for k, v in (("ART", round(now["art"], 4)), ("FMS", round(fmsv, 3)), ("DFTHIN", round(now["thin"], 1)),
                 ("DFEXT", round(now["ext"], 1)), ("DFSCC", round(now["scc"], 1)), ("DFOTHER", round(now["other"], 1)),
                 ("DFTOTAL", round(now["total"], 1)), ("POF", gff * fmsv * now["total"]), ("POFCATNOW", pc_now),
                 ("DFPLAN", round(end["total"], 1)), ("POFCAT", pc_end), ("COFCOST", round(fc, 2)), ("COFFIN", cfin),
                 ("COFAREA", carea), ("COFCAT", cof), ("FINRISK", round(p["risk_now"], 2)),
                 ("RISKCELLNOW", "%d%s" % (pc_now, cof)), ("FINRISKPLAN", round(p["risk_plan"], 2)),
                 ("RISKCELL", "%d%s" % (pc_end, cof)), ("RISK", risk_level(pc_end, cof)), ("DRIVER", p["driver"]),
                 ("INSPEFFREQ", p["need"]), ("FINRISKMIT", round(p["risk_mit"], 2)),
                 ("INSPMETHOD", (METHOD.get(p["driver"], "") + " (effectiveness %s or better)" % p["need"])[:100])):
        setv(a, k, v)
    setv(a, "TARGETDATE", add_years(rbi, p["t_target"]) if p["t_target"] is not None else None)
    nxt = add_years(rbi, p["due"])
    setv(a, "NEXTINSPDATE", nxt)
    setv(a, "INSPINTERVAL", round(max(years(last_insp, nxt), 0.0), 1))

    note = ("Thinning: Cr = max(LT %.3f, ST %.3f) = %.3f mm/y, t_rdi %.2f mm, Art %.4f, SR %.3f, confidence %s, "
            "remaining life %.1f y. Df now: thin %.1f, ext/CUI %.1f, cracking %.1f, HTHA/fatigue %.1f -> total %.1f "
            "(PoF cat %d); at plan date %.1f (cat %d). FMS %.3f, PoF %.2e /y. FC %.0f USD (%s), area %.0f m2 (%s) -> CoF %s. "
            "Risk %.0f USD/y now, %.0f at plan date vs target %.0f. Inspection due in %.1f y (%s); "
            "%s inspection of effectiveness %s brings plan-date risk to %.0f USD/y. %s"
            % (crlt, crst, cr, tlast, now["art"], sr, eq["conf"], rl, now["thin"], now["ext"], now["scc"], now["other"],
               now["total"], pc_now, end["total"], pc_end, fmsv, gff * fmsv * now["total"], fc, cfin,
               a.getDouble("CONSAREA"), carea, cof, p["risk_now"], p["risk_plan"], target, p["due"], p["why"],
               p["driver"], p["need"], p["risk_mit"], " ".join(notes)))
    setv(a, "CALCNOTE", note[:1000])

if not _frozen:
    _run()
