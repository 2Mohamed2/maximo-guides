@Override
public void execute(TririgaWS tws, HttpServletRequest req, HttpServletResponse res) throws ServletException, IOException {
    res.setContentType("application/json; charset=UTF-8");
    res.setHeader("Cache-Control", "no-store");
    long t0 = System.currentTimeMillis();
    String action = req.getParameter("action") == null ? "me" : req.getParameter("action");
    String body;
    try {
        switch (action) {
            case "me" -> body = me(req);
            case "buildings" -> body = buildings(tws);
            case "floors" -> body = floors(tws, text(req.getParameter("building")));
            case "spaces" -> body = spaces(tws, text(req.getParameter("building")), text(req.getParameter("floor")));
            default -> { res.setStatus(400); body = "{\"error\":\"unknown action\"}"; }
        }
    } catch (IllegalArgumentException e) {
        res.setStatus(400);
        body = "{\"error\":" + q(e.getMessage()) + "}";
    } catch (Exception e) {
        res.setStatus(500);
        body = "{\"error\":" + q(e.getClass().getSimpleName() + ": " + e.getMessage()) + "}";
    }
    long ms = System.currentTimeMillis() - t0;
    res.getWriter().write(body.substring(0, body.length() - 1) + (body.length() > 2 ? "," : "") + "\"ms\":" + ms + "}");
}

/** Active buildings: one dynamic query, filtered on the status, paged by the engine. */
private String buildings(TririgaWS tws) throws Exception {
    DisplayLabel[] cols = {
        new DisplayLabel("triIdTX", "id", "General"),
        new DisplayLabel("triNameTX", "name", "General"),
        new DisplayLabel("triCityTX", "city", "General"),
        new DisplayLabel("triPrimaryUseCL", "use", "General")};
    Filter[] where = {filter("triStatusCL", "Active")};
    QueryResult r = tws.runDynamicQuery(null, "Location", new String[]{"triBuilding"}, null, null, null,
        COMPANY, cols, null, null, where, null, 1, 500);
    return "{\"total\":" + r.getTotalResults() + ",\"rows\":" + rows(r, cols) + "}";
}

private static Filter filter(String field, String value) {
    Filter f = new Filter();
    f.setSectionName("General");
    f.setFieldName(field);
    f.setDataType(STRING);
    f.setOperator(EQUALS);
    f.setValue(value);
    return f;
}

/** The rows of a query result as JSON objects keyed by the labels we asked for. */
private static String rows(QueryResult r, DisplayLabel[] cols) {
    List<String> out = new ArrayList<>();
    QueryResponseHelper[] hs = r == null ? null : r.getQueryResponseHelpers();
    if (hs != null) {
        for (QueryResponseHelper h : hs) {
            StringBuilder o = new StringBuilder("{\"recordId\":").append(h.getRecordId());
            QueryResponseColumn[] cs = h.getQueryResponseColumns();
            for (int i = 0; cs != null && i < cs.length; i++) {
                String key = cs[i].getLabel() != null ? cs[i].getLabel() : cols[Math.min(i, cols.length - 1)].getLabel();
                String val = cs[i].getDisplayValue() != null ? cs[i].getDisplayValue() : cs[i].getValue();
                o.append(',').append(q(key)).append(':').append(q(val == null ? "" : val));
                // Numbers with a unit (area, cost) come back in the signed-in user's unit: pass the unit on.
                if (cs[i].getUom() != null && !cs[i].getUom().isBlank())
                    o.append(',').append(q(key + "Unit")).append(':').append(q(cs[i].getUom()));
            }
            out.add(o.append('}').toString());
        }
    }
    return "[" + String.join(",", out) + "]";
}

        subprocess.run(["javac", "--release", "21", "-nowarn", "-cp", os.pathsep.join(map(str, CP)), "-d", out, *sources], check=True)
        jar = DIST / "LocationReader.jar"
        with zipfile.ZipFile(jar, "w") as z:
            add(z, "META-INF/MANIFEST.MF", b"Manifest-Version: 1.0\r\n\r\n")
            for p in sorted(Path(out).rglob("*.class")):
                add(z, p.relative_to(out).as_posix(), p.read_bytes())
    web = DIST / "LocationReader_Web.zip"
    with zipfile.ZipFile(web, "w") as z:
        for p in sorted((HERE / "web").rglob("*")):
            if p.is_file():
                add(z, p.relative_to(HERE / "web").as_posix(), p.read_bytes())
    for f in (jar, web):
        print(f"{f.name}: {f.stat().st_size:,} bytes, {zipfile.ZipFile(f).namelist()}")

async function get(params, callEl) {
  const qs = new URLSearchParams(params).toString();
  const r = await fetch(API + '?' + qs, {credentials:'same-origin'});
  const j = await r.json();
  $(callEl).textContent = 'GET ' + API + '?' + qs + '  →  HTTP ' + r.status + ', ' + j.ms + ' ms';
  if (!r.ok) throw new Error(j.error || r.status);
  return j;
}

/**
 * Floors of a building, each with its number of active spaces. Two queries filtered on the parent building's
 * name (a text field every location carries), counted here. getChildren is not used: inside a Class Loader its
 * answer came back as HTTP 401 "Login Failed." although the data had been read.
 */
private String floors(TririgaWS tws, String building) throws Exception {
    DisplayLabel[] fcols = {new DisplayLabel("triNameTX", "name", "General")};
    QueryResult fr = tws.runDynamicQuery(null, "Location", new String[]{"triFloor"}, null, null, null,
        COMPANY, fcols, null, null, new Filter[]{filter("triParentBuildingTX", building)}, null, 1, 500);
    DisplayLabel[] scols = {new DisplayLabel("triParentFloorTX", "floor", "General")};
    Map<String, Integer> perFloor = new HashMap<>();
    QueryResult sr = tws.runDynamicQuery(null, "Location", new String[]{"triSpace"}, null, null, null,
        COMPANY, scols, null, null, new Filter[]{filter("triParentBuildingTX", building), filter("triStatusCL", "Active")}, null, 1, 1000);
    while (sr != null && sr.getQueryResponseHelpers() != null && sr.getQueryResponseHelpers().length > 0) {
        for (QueryResponseHelper h : sr.getQueryResponseHelpers()) {
            QueryResponseColumn[] cs = h.getQueryResponseColumns();
            String f = cs == null || cs.length == 0 ? "" : cs[0].getDisplayValue();
            perFloor.merge(f == null ? "" : f, 1, Integer::sum);
        }
        if (sr.getContinuationToken() == null) break;
        sr = tws.runDynamicQueryContinue(sr.getContinuationToken());
    }
    List<String> out = new ArrayList<>();
    QueryResponseHelper[] hs = fr == null ? null : fr.getQueryResponseHelpers();
    if (hs != null) {
        for (QueryResponseHelper h : hs) {
            QueryResponseColumn[] cs = h.getQueryResponseColumns();
            String name = cs == null || cs.length == 0 ? "" : cs[0].getDisplayValue();
            out.add("{\"recordId\":" + h.getRecordId() + ",\"name\":" + q(name == null ? "" : name)
                + ",\"spaces\":" + perFloor.getOrDefault(name, 0) + "}");
        }
    }
    return "{\"total\":" + (fr == null ? 0 : fr.getTotalResults()) + ",\"rows\":[" + String.join(",", out) + "]}";
}
