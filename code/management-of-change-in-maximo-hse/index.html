select r.riskassessnum, r.status, r.location, h.hazardid
from workorder w
join plusgriskassess r on r.plusgriskassessid = w.plusgriskassessid
join plusghazrisk z on z.plusgriskassessid = r.plusgriskassessid
join hazard h on h.hazarduid = z.hazarduid
where w.wonum = '1379'

select 'Reviewer', reviewer, reviewed, convert(varchar(16), signoffdate, 120), comments
from plusgmocrvlst
where wonum = '1379'
union all select 'Approver', approver, approved, convert(varchar(16), signoffdate, 120), comments
from plusgmocapprlist
where wonum = '1379'
union all select 'Authorizer', authorizer, authorized, convert(varchar(16), signoffdate, 120), comments
from plusgmocauthlist
where wonum = '1379'

from plusgmocpostlist
where wonum = '1379'

from plusgmocrevact
where workorderid = 134067
union all select 'Closure', stdactnum, actionby, completed
from plusgmocclosurelist

select c.trainingnum, c.isrequired, p.person, p.iscomplete
from plusgmoccourse c
join plusgmoccosper p on p.wonum = c.wonum and p.trainingnum = c.trainingnum
where c.wonum = '1379'
order by p.person
