def upload(m, type_uuid, path):
    lib = f"/api/v2/core/deviceTypes/{type_uuid}/images"
    have = m.call("GET", lib + "?includeDevices=false", ok_fail=True)
    for im in (have if isinstance(have, list) else []):
        if im.get("fileName") == path.name:
            return str(im.get("id") or im.get("fileName"))
    bnd = "----fgline" + _uuid.uuid4().hex
    body = (f"--{bnd}\r\nContent-Disposition: form-data; name=\"file\"; filename=\"{path.name}\"\r\nContent-Type: image/jpeg\r\n\r\n").encode() \
        + path.read_bytes() + f"\r\n--{bnd}--\r\n".encode()
    h = {k: v for k, v in m.h.items() if k.lower() != "content-type"}
    h["Content-Type"] = f"multipart/form-data; boundary={bnd}"
    req = urllib.request.Request(m.base + lib, data=body, method="POST", headers=h)
    with urllib.request.urlopen(req, context=ssl._create_unverified_context(), timeout=120) as f:
        r = json.loads(f.read() or b"{}")
    r = r[0] if isinstance(r, list) and r else r
    return str(r.get("id") or r.get("fileName") or path.name)

    (2 * N2_FLOW / (N2_FLOW + H2_FLOW)).round(2)

AIR_INGRESS  (FG_TINBATH, PythonExpression)
    (
      (ATM_O2 - 2 * N2_FLOW / (N2_FLOW + H2_FLOW))
      * (N2_FLOW + H2_FLOW)
      / 209000
      * 1000
      / 60
    ).clip(lower=0).round(2)

    (25 * (N2_FLOW + H2_FLOW) / 1850).round(1)

def rnd(*k):
    return int(hashlib.md5("|".join(map(str, k)).encode()).hexdigest()[:8], 16) / 0xFFFFFFFF


def noise(key, t, amp, period=300):
    """Smooth noise in -amp..amp: stable random points every `period` seconds, interpolated."""
    i, f = divmod(t / period, 1)
    a, b = rnd(key, int(i)) - .5, rnd(key, int(i) + 1) - .5
    return 2 * amp * (a + (b - a) * (3 * f * f - 2 * f * f * f))


def hour(t):                                          # plant local time (UTC+4)
    d = datetime.fromtimestamp(t, timezone.utc)
    return (d.hour + 4) % 24 + d.minute / 60


def window(t, start_s, dur_s, rise_s=1800.0, fall_s=3600.0):
    """0 outside, 1 inside, smooth edges: an event from start_s lasting dur_s."""
    if t < start_s - 1 or t > start_s + dur_s + fall_s:
        return 0.0
    up = min(1.0, max(0.0, (t - start_s) / rise_s))
    down = min(1.0, max(0.0, 1 - (t - start_s - dur_s) / fall_s))
    x = min(up, down)
    return x * x * (3 - 2 * x)

    return max(0.0, min(1.0, (t - s["since"]) / 60 / max(1, s.get("ramp_min", 10))))

    tind = max(0.0, 0.05 + noise("tin", t, .02, 900) + 0.62 * tin)

class State:
    lock = threading.Lock()
    scenarios, latest, published, path = {}, {}, 0, None

    @classmethod
    def set(cls, name, active, ramp_min=None):
        with cls.lock:
            s = cls.scenarios.setdefault(name, {})
            s["active"] = bool(active)
            if ramp_min:
                s["ramp_min"] = max(1, min(120, int(ramp_min)))
            s["since"] = time.time() if active else None
            if cls.path:
                Path(cls.path).write_text(json.dumps(cls.scenarios), encoding="utf-8")
            return dict(cls.scenarios)


class Http(BaseHTTPRequestHandler):
    def log_message(self, *a):
        pass

    def send(self, code, body, ctype="application/json"):
        b = body if isinstance(body, bytes) else json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(b)

    def do_GET(self):
        if self.path in ("/", "/index.html"):
            return self.send(200, (HERE / "page.html").read_bytes(), "text/html; charset=utf-8")
        if self.path == "/api/latest":
            with State.lock:
                return self.send(200, {"ts": time.time(), "published": State.published, "scenarios": State.scenarios,
                                       "devices": State.latest})
        if self.path == "/healthz":
            return self.send(200, {"ok": True})
        self.send(404, {"error": "not found"})

    def do_POST(self):
        if self.path != "/api/scenario":
            return self.send(404, {"error": "not found"})
        key = os.environ.get("DEMO_KEY", "")
        if not key or self.headers.get("X-Demo-Key") != key:
            return self.send(403, {"error": "demo key required"})
        b = json.loads(self.rfile.read(int(self.headers.get("Content-Length", 0)) or 0) or b"{}")
        if b.get("name") not in SCEN_NAMES:
            return self.send(400, {"error": "unknown scenario"})
        self.send(200, State.set(b["name"], b.get("active"), b.get("ramp_min")))

  O2_expected   = c_N2 * N2 / (N2 + H2)                    c_N2 = N2 supply spec (2 ppm)
  air ingress   = (O2 - O2_expected) * (N2 + H2) / 209,000  Nm3/h -> L/min (x 1000/60), if all excess O2 is air
  P_expected    = P_design * (N2 + H2) / (N2_design + H2_design)
  A leak raises O2 AND lowers the pressure AND wets the gas; a contaminated N2 supply raises O2 only.
Ribbon mass balance:
  glass carried (t/day) = speed (m/min) x width (m) x thickness (mm) x density (t/m3) x 1.44
  mass gap %            = weigher / glass carried - 1
"""


def _k(twin):
    a, r = twin["design"]["atmosphere"], twin["design"]["ribbon"]
    return dict(c=a["n2_supply_o2_spec_ppm"], air=a["air_o2_ppm"], p0=a["design_pressure_pa"],
                q0=a["n2_design_nm3h"] + a["h2_design_nm3h"], rho=r["density_t_m3"], dp0=a["dew_point_design_c"])


def _parts(twin):
    k = _k(twin)
    Q = "(df['N2_FLOW_NM3H'] + df['H2_FLOW_NM3H']).clip(lower=1)"

    EX = f"(df['ATM_O2_PPM'] - {O2E})"
    AIR = f"({EX} * {Q} / {k['air']} * 1000 / 60).clip(lower=0)"
    PE = f"({k['p0']} * {Q} / {k['q0']})"
    PD = f"({PE} - df['BATH_PRESSURE_PA'])"
    RT = f"(df['RIBBON_SPEED_M_MIN'] * df['RIBBON_WIDTH_M'] * df['THICKNESS_MM'] * {k['rho']} * 1.44)"
    GAP = f"(100 * (df['WEIGHER_TPD'] / {RT}.clip(lower=1) - 1))"

"""Physics of the FG-L1-TB tin bath and its ribbon: the stand-in for the real line (it sends only sensor values).

Per step (dt seconds):
  gas        N2 at design flow; H2 trimmed up by the operator when O2 rises (reducing margin)
  ingress    side seal leak: air in at up to 2.6 L/min; the opening lowers the over-pressure; air brings O2 and moisture
  supply     contaminated N2: the supply O2 rises from 2 ppm to ~20 ppm; pressure and dew point unchanged
  O2         first-order mixing toward (c_N2 * N2 + 209,000 * Q_air) / (N2 + H2 + Q_air), time constant ~3 min
  ribbon     600 t/day at 4 mm on a 3.8 m ribbon; speed follows from the mass balance; the weigher reads the batch
             (calibration drift adds up to +1.5 %)
  defects    tin defects follow the bath O2 with a few minutes' lag (oxidised tin picked up by the ribbon)
"""
import math
import random


class Fault:
    def __init__(self):
        self.sev, self.target, self.rate = 0.0, 0.0, 0.0

    def set(self, target, ramp_s):
        self.target = max(0.0, min(1.0, target))
        self.rate = abs(self.target - self.sev) / max(ramp_s, 1.0)

    def step(self, dt):
        if self.sev < self.target:
            self.sev = min(self.target, self.sev + self.rate * dt)
        elif self.sev > self.target:
            self.sev = max(self.target, self.sev - self.rate * dt)


class Plant:
    def __init__(self, twin, seed=601):
        self.twin, self.rng = twin, random.Random(seed)
        self.d = twin["design"]
        self.faults = {k: Fault() for k in twin["failure_modes"] if not k.startswith("$")}
        self.t = 0.0
        self.running, self.trip = True, 0
        a = self.d["atmosphere"]
        self.o2 = a["n2_supply_o2_spec_ppm"] * a["n2_design_nm3h"] / (a["n2_design_nm3h"] + a["h2_design_nm3h"])
        self.dew = a["dew_point_design_c"]
        self.tin = 0.05
        self.h2_trim = 0.0
        self.th = 4.0

    def set_fault(self, name, severity, ramp_s):
        self.faults[name].set(severity, ramp_s)

    def clear_faults(self, ramp_s=5):
        for f in self.faults.values():
            f.set(0.0, ramp_s)

    def reset(self):
        self.trip, self.running = 0, True

    def fault_state(self):
        return {k: round(v.sev, 3) for k, v in self.faults.items()}

    def _n(self, key, amp, period):
        """smooth noise, stable per key"""
        i, f = divmod(self.t / period, 1)
        a = random.Random(f"{key}{int(i)}").random() - .5
        b = random.Random(f"{key}{int(i) + 1}").random() - .5
        return 2 * amp * (a + (b - a) * (3 * f * f - 2 * f ** 3))

    def step(self, dt):
        self.t += dt
        for f in self.faults.values():
            f.step(dt)
        a, rb = self.d["atmosphere"], self.d["ribbon"]
        leak, sup, drift = (self.faults[k].sev for k in ("side_seal_leak", "n2_supply_o2", "weigher_drift"))
        n2 = a["n2_design_nm3h"] + self._n("n2", 12, 600)
        air_lpm = 2.6 * leak
        q_air = air_lpm * 60 / 1000
        c_n2 = a["n2_supply_o2_spec_ppm"] + 18.0 * sup
        # operator / controller trims H2 up as O2 rises (keeps the atmosphere reducing)
        target_trim = max(0.0, 2.4 * (self.o2 - 2.0))
        self.h2_trim += (target_trim - self.h2_trim) * min(1.0, dt / 240)
        h2 = a["h2_design_nm3h"] + self.h2_trim + self._n("h2", 1.5, 600)
        q = n2 + h2 + q_air
        o2_eq = (c_n2 * n2 + a["air_o2_ppm"] * q_air) / q
        self.o2 += (o2_eq - self.o2) * min(1.0, dt / 180)
        dew_eq = a["dew_point_design_c"] + 16.0 * math.sqrt(air_lpm)
        self.dew += (dew_eq - self.dew) * min(1.0, dt / 240)
        self.tin += ((0.05 + 0.034 * max(0.0, self.o2 - 2.0)) - self.tin) * min(1.0, dt / 300)
        p = a["design_pressure_pa"] * (n2 + h2) / (a["n2_design_nm3h"] + a["h2_design_nm3h"]) - 4.3 * air_lpm
        pull = rb["pull_tpd"] + self._n("pull", 5, 1800)
        width = rb["gross_width_m"] + self._n("w", 0.004, 300)
        speed = pull / (1.44 * rb["density_t_m3"] * self.th * width)
        return {"n2": n2, "h2": h2, "o2": self.o2, "h2pct": 100 * h2 / (n2 + h2), "dew": self.dew, "p": p,
                "lip": 1100 + self._n("lip", 2, 600), "entry": 1042 + self._n("ent", 3, 600), "exit": 612 + self._n("ext", 2.5, 600),
                "speed": speed, "width": width, "th": self.th, "weigher": pull * (1 + 0.015 * drift), "tin": self.tin}

    def measure(self, x):
        g = lambda v, s: v + self.rng.gauss(0, s)
        return {"n2_flow_nm3h": round(g(x["n2"], 3), 1), "h2_flow_nm3h": round(g(x["h2"], 0.5), 1),
                "atm_o2_ppm": round(max(0.1, g(x["o2"], 0.12)), 2), "atm_h2_pct": round(g(x["h2pct"], 0.03), 2),
                "dew_point_c": round(g(x["dew"], 0.25), 1), "bath_pressure_pa": round(g(x["p"], 0.2), 1),
                "lip_temp_c": round(g(x["lip"], 0.8), 1), "entry_temp_c": round(g(x["entry"], 0.8), 1), "exit_temp_c": round(g(x["exit"], 0.8), 1),
                "ribbon_speed_m_min": round(g(x["speed"], 0.005), 3), "ribbon_width_m": round(g(x["width"], 0.002), 3),
                "thickness_mm": round(g(x["th"], 0.008), 3), "thickness_sp_mm": x["th"],
                "weigher_tpd": round(g(x["weigher"], 1.0), 1), "tin_defects_10m2": round(max(0.0, g(x["tin"], 0.008)), 3)}
