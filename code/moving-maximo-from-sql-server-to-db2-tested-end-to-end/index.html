"""Find SQL Server-only SQL stored inside a Maximo database (run against the SOURCE SQL Server database).

Created by Mohamed Chihaoui.

Same idea as find_oracle_sql.py, for Maximo on Microsoft SQL Server. SQL Server 2022 has no regular
expressions, so it works in two steps:

  python find_tsql.py extract > extract.sql
  sqlcmd -C -S <host> -U maximo -P <password> -d <database> -y 0 -s "~" -i extract.sql -o extract.txt
  python find_tsql.py classify extract.txt > findings.csv

Classes:
  FAILS   - Db2 11.5 rejects it (tested on Db2 11.5.8)
  SILENT  - Db2 runs it and can return different rows. On SQL Server, Maximo databases normally use a
            case- and accent-insensitive collation (here Latin1_General_CI_AI): 'appr' finds APPR.
            On Db2 the comparison is exact.
Output (classify): class~rule~table.column~record key~first 300 characters~what to do on Db2
"""
import re
import sys

# (table, readable record key as T-SQL, SQL-bearing column)
SOURCES = [
    ("QUERY", "APP+':'+CLAUSENAME", "CLAUSE"),
    ("MAXRELATIONSHIP", "PARENT+'.'+NAME", "WHERECLAUSE"),
    ("CONDITION", "CONDITIONNUM", "EXPRESSION"),
    ("ESCALATION", "ESCALATION", "CONDITION"),
    ("WFCONDITION", "PROCESSNAME+':'+CAST(NODEID AS VARCHAR(10))", "CONDITION"),
    ("WFACTION", "PROCESSNAME+':'+CAST(ACTIONID AS VARCHAR(10))", "CONDITION"),
    ("WFASSIGNMENT", "PROCESSNAME+':'+CAST(NODEID AS VARCHAR(10))+':'+CAST(ASSIGNID AS VARCHAR(10))", "CONDITION"),
    ("KPIMAIN", "KPINAME", "CLAUSE"),
    ("KPIMAIN", "KPINAME", "SELECTSTMT"),
    ("MAXINTOBJECT", "INTOBJECTNAME", "RESTRICTWHERE"),
    ("MAXIFACEEXPORT", "IFACENAME", "EXPORTWHERE"),
    ("OSLCQUERY", "INTOBJECTNAME+':'+CLAUSENAME", "CLAUSE"),
    ("OSLCINTQUERY", "PROVIDERNAME+':'+QUERYNUM", "WHERECLAUSE"),
    ("MAXTABLEDOMAIN", "DOMAINID+':'+OBJECTNAME", "VALIDTNWHERECLAUSE"),
    ("MAXTABLEDOMAIN", "DOMAINID+':'+OBJECTNAME", "LISTWHERECLAUSE"),
    ("MAXDYNAMICDOMLINK", "OBJECTNAME+'.'+ATTRIBUTENAME", "DOMAINWHERE"),
    ("MAXVIEW", "VIEWNAME", "VIEWWHERE"),
    ("MAXVIEW", "VIEWNAME", "VIEWSELECT"),
    ("SCRIPTLAUNCHPOINT", "LAUNCHPOINTNAME", "CONDITION"),
    ("SLA", "SLANUM", "CONDITION"),
    ("REPORTADHOC", "REPORTNAME", "WHERECLAUSE"),
    ("PLUSGSEQOFEVENTS", "TICKETID+':'+NAME", "CONDITION"),
    ("SKDQUERY", "OBJECTNAME+':'+QUERYNAME", "WHERECLAUSE"),
    ("AUTOSCRIPT", "AUTOSCRIPT", "SOURCE"),
    ("REPORTDESIGN", "REPORTNAME", "DESIGN"),
]
LONG = {("AUTOSCRIPT", "SOURCE"), ("REPORTDESIGN", "DESIGN")}   # scripts and BIRT XML: the noisy rules are skipped

W = r"(^|[^a-z0-9_.])"   # word start
RULES = [
    ("TOP n", "FAILS", W + r"top\s*\(?\s*[0-9]", "FETCH FIRST n ROWS ONLY"),
    ("GETDATE / SYSDATETIME", "FAILS", W + r"(getdate|getutcdate|sysdatetime|sysutcdatetime)\s*\(", "CURRENT TIMESTAMP"),
    ("DATEADD", "FAILS", W + r"dateadd\s*\(", "labelled duration: d - 7 DAYS, d + 3 MONTHS"),
    ("DATEDIFF", "FAILS", W + r"datediff\s*\(", "DAYS(b) - DAYS(a), or TIMESTAMPDIFF"),
    ("DATEPART", "FAILS", W + r"datepart\s*\(", "YEAR(d), MONTH(d), DAY(d), HOUR(d)"),
    ("LEN", "FAILS", W + r"len\s*\(", "LENGTH(RTRIM(x)) (LEN ignores trailing blanks)"),
    ("CHARINDEX", "FAILS", W + r"charindex\s*\(", "LOCATE(find, s) or INSTR(s, find)"),
    ("CONVERT", "FAILS", W + r"convert\s*\(", "CAST, VARCHAR_FORMAT(d, 'YYYY-MM-DD')"),
    ("IIF / STRING_AGG / NEWID / EOMONTH / FORMAT", "FAILS", W + r"(iif|string_agg|newid|eomonth|format)\s*\(",
     "CASE, LISTAGG, GENERATE_UNIQUE(), LAST_DAY, VARCHAR_FORMAT"),
    # Maximo's own :&synonymlist&_DOMAIN[VALUE] syntax is not a bracket identifier
    ("[bracket] identifier", "FAILS", r"(^|[^a-z0-9_&'])\[[a-z_][a-z0-9_ ]*\]", "plain or double-quoted identifier"),
    ("dbo. prefix", "FAILS", W + r"dbo\.", "drop the schema prefix (Maximo sets the current schema)"),
    ("NOLOCK hint", "FAILS", r"(with\s*\(\s*nolock|\bnolock\b)", "remove the hint (Db2 Maximo uses currently committed reads)"),
    ("string + string", "FAILS", r"('\s*\+|\+\s*')", "|| or CONCAT(); + on strings is SQL0420N"),
    ("LIKE '[...]' range", "SILENT", r"like\s+n?'[^']*\[[^\]]+\][^']*'", "SQL Server reads [A-C] as a range, Db2 as text: use OR / BETWEEN"),
    # matched on the ORIGINAL text: a literal holding a lowercase letter, compared to a column
    ("lowercase literal compare", "SILENT", r"(?i:=|<>|!=|\bin\s*\(|like)\s*[nN]?'[^']*[a-z][^']*'",
     "SQL Server compares without case or accents, Db2 exactly: match the stored case or wrap with UPPER()"),
]
SKIP_LONG = {"lowercase literal compare", "string + string", "[bracket] identifier"}

def extract():
    """One result set; tables of add-ons that are not installed are skipped (dynamic SQL per table)."""
    parts = []
    for table, key, col in SOURCES:
        q = (f"select '{table}.{col}', cast({key} as nvarchar(200)), "
             f"replace(replace(replace(cast({col} as nvarchar(max)), char(13), ' '), char(10), ' '), '~', '-') "
             f"from dbo.{table.lower()} where {col} is not null").replace("'", "''")
        parts.append(f"if object_id('dbo.{table.lower()}') is not null insert into #f exec('{q}');")
    return ("-- Generated by find_tsql.py (Created by Mohamed Chihaoui)\nset nocount on;\n"
            "create table #f (w nvarchar(100), k nvarchar(200), t nvarchar(max));\n"
            + "\n".join(parts) + "\nselect w, k, t from #f order by w, k;\n")

def classify(path):
    out = []
    for line in open(path, encoding="utf-8", errors="replace"):
        p = line.rstrip("\n").split("~", 2)
        if len(p) < 3:
            continue
        where, key, text = p
        low = text.lower()
        tc = tuple(where.split("."))
        for rid, cls, rx, fix in RULES:
            if tc in LONG and rid in SKIP_LONG:
                continue
            if re.search(rx, text if rid == "lowercase literal compare" else low):
                out.append("~".join([cls, rid, where, key.strip(), text[:300], fix]))
    out.sort()
    print("\n".join(out))

if __name__ == "__main__":
    if sys.argv[1] == "extract":
        sys.stdout.write(extract())
    else:
        classify(sys.argv[2])

// CvtTool - checks and fixes after a Maximo Oracle-to-DB2 move with Unlcvt + maxinst.
// Created by Mohamed Chihaoui.
//
//   java CvtTool.java counts   -> row count of every Maximo table on both sides (counts.csv)
//   java CvtTool.java blobs    -> copy the binary values Unlcvt leaves out (images, BIRT resources,
//                                 geometry, report output, encrypted fields); source Oracle or SQL Server
//   java CvtTool.java empties  -> count '' values in text columns on Db2 (and on a SQL Server source)
//   java CvtTool.java restore out.csv T1,T2 -> replace the Db2 rows of these tables with the source rows.
//                                 maxinst runs IBM's VCATCHUPMAXINST.dbc, which deletes the Assistant tool
//                                 definitions (MAXINTOBJTOOL) of four object structures and every MAXAICFGRAG row,
//                                 and reinserts IBM's defaults: customised or added tools are lost without it.
//
// Connections come from environment variables:
//   SRC_URL SRC_USER SRC_PASS SRC_SCHEMA   (Oracle or SQL Server)
//   TGT_URL TGT_USER TGT_PASS TGT_SCHEMA   (DB2)
// Run with the Maximo JDBC drivers on the class path:
//   java -cp oraclethin.jar:sqljdbc.jar:db2jcc.jar CvtTool.java counts
import java.io.*;
import java.sql.*;
import java.util.*;

public class CvtTool {
    static String env(String k) { String v = System.getenv(k); if (v == null) throw new IllegalArgumentException(k + " not set"); return v; }
    static Connection src() throws SQLException { return DriverManager.getConnection(env("SRC_URL"), env("SRC_USER"), env("SRC_PASS")); }
    static Connection tgt() throws SQLException { return DriverManager.getConnection(env("TGT_URL"), env("TGT_USER"), env("TGT_PASS")); }

    public static void main(String[] a) throws Exception {
        String out = a.length > 1 ? a[1] : "/work/" + a[0] + ".csv";
        switch (a[0]) {
            case "counts" -> counts(out);
            case "blobs" -> blobs(out);
            case "empties" -> empties(out, a.length > 2 ? a[2] : "WORKORDER,ASSET,LOCATIONS,PERSON,LABOR,ITEM,INVENTORY,PO,POLINE,PR,TICKET,JOBPLAN,PM,MATUSETRANS,COMPANIES");
            case "restore" -> restore(out, a.length > 2 ? a[2] : "MAXINTOBJTOOL,MAXINTOBJTOOLPARAM,MAXAICFGRAG");
            default -> throw new IllegalArgumentException("counts | blobs | empties | restore");
        }
    }

    // Persistent Maximo tables (not views), from the source data dictionary
    static List<String[]> tables(Connection s) throws SQLException {
        List<String[]> l = new ArrayList<>();
        try (ResultSet r = s.createStatement().executeQuery(
                "select t.tablename, t.uniquecolumnname from maxtable t join maxobject o on o.objectname = t.tablename "
                + "where o.persistent = 1 and o.isview = 0 order by 1")) {
            while (r.next()) l.add(new String[]{r.getString(1), r.getString(2)});
        }
        return l;
    }

    static long count(Connection c, String sql) {
        // close the statement too: Db2 runs out of JDBC packages (SQL0805N) otherwise
        // views can be large joins: give each count 60 s (-2 = timed out)
        try (Statement st = c.createStatement()) {
            st.setQueryTimeout(60);
            try (ResultSet r = st.executeQuery(sql)) { r.next(); return r.getLong(1); }
        } catch (SQLTimeoutException e) { return -2; }
        catch (SQLException e) { return e.getMessage() != null && e.getMessage().toLowerCase().contains("timeout") ? -2 : -1; }
    }

    // Row counts of every persistent table, then of every Maximo view: a view can return other rows on Db2
    // even when its tables match (SQL Server joins text without regard to case, Db2 does not).
    static void counts(String out) throws Exception {
        try (Connection s = src(); Connection t = tgt(); PrintWriter w = new PrintWriter(out, "UTF-8")) {
            String ts = env("TGT_SCHEMA");
            w.println("object,kind,source_rows,db2_rows,verdict");
            int same = 0, diff = 0, vsame = 0, vdiff = 0;
            for (String[] tb : tables(s)) {
                long o = count(s, "select count(*) from " + tb[0]);
                long d = count(t, "select count(*) from " + ts + "." + tb[0]);
                String v = o == d ? "SAME" : (d < 0 ? "MISSING ON DB2" : "DIFFERENT");
                if (o == d) same++; else diff++;
                w.println(tb[0] + ",table," + o + "," + d + "," + v);
            }
            List<String> views = new ArrayList<>();
            try (Statement st = s.createStatement(); ResultSet r = st.executeQuery(
                    "select viewname from maxview where viewname in (select objectname from maxobject where isview = 1) order by 1")) {
                while (r.next()) views.add(r.getString(1));
            }
            for (String vw : views) {
                long o = count(s, "select count(*) from " + vw);
                long d = count(t, "select count(*) from " + ts + "." + vw);
                if (o == -1 && d == -1) continue;
                String v = (o == -2 || d == -2) ? "TIMEOUT" : o == d ? "SAME" : (d < 0 ? "ERROR ON DB2" : o < 0 ? "ERROR ON SOURCE" : "DIFFERENT");
                if (o == d) vsame++; else vdiff++;
                w.println(vw + ",view," + o + "," + d + "," + v);
            }
            System.out.println("tables same=" + same + " different=" + diff + "; views same=" + vsame + " different=" + vdiff + " -> " + out);
        }
    }

    // Tables whose ids maxinst renumbers: match on the record they belong to instead of the id.
    static final Map<String, String[]> MATCH_ON = Map.of("IMGLIB", new String[]{"REFOBJECT", "REFOBJECTID"});
    // Rebuilt by the Manage build on the new database: redeploy customised mobile apps, do not copy.
    static final Set<String> SKIP = Set.of("MAFAPPDATA");

    static void blobs(String out) throws Exception {
        try (Connection s = src(); Connection t = tgt(); PrintWriter w = new PrintWriter(out, "UTF-8")) {
            t.setAutoCommit(false);
            String ts = env("TGT_SCHEMA");
            Map<String, String> keyOf = new HashMap<>();
            for (String[] tb : tables(s)) keyOf.put(tb[0], tb[1]);
            // binary columns of the source: Oracle BLOB / RAW (encrypted CRYPTO fields), SQL Server varbinary / image
            boolean ora = s.getMetaData().getDatabaseProductName().toLowerCase().contains("oracle");
            String colSql = ora
                ? "select table_name, column_name, data_type from user_tab_columns where data_type in ('BLOB','RAW','LONG RAW') order by 1, 2"
                : "select upper(table_name), upper(column_name), data_type from information_schema.columns "
                  + "where data_type in ('varbinary','image','binary') order by 1, 2";
            List<String[]> cols = new ArrayList<>();
            try (Statement st = s.createStatement(); ResultSet r = st.executeQuery(colSql)) {
                while (r.next()) if (keyOf.containsKey(r.getString(1))) cols.add(new String[]{r.getString(1), r.getString(2), r.getString(3)});
            }
            w.println("table,column,matched_on,rows_with_bytes,copied_to_db2,not_found_on_db2,bytes,source_type");
            for (String[] c : cols) {
                String tb = c[0], col = c[1];
                String[] match = MATCH_ON.getOrDefault(tb, new String[]{keyOf.get(tb)});
                String sel = String.join(", ", match), where = String.join(" = ? and ", match) + " = ?";
                int found = 0, upd = 0, miss = 0; long bytes = 0;
                try (Statement st = s.createStatement();
                     ResultSet r = st.executeQuery("select " + sel + ", " + col + " from " + tb + " where " + col + " is not null"
                         + (ora ? (c[2].equals("BLOB") ? " and dbms_lob.getlength(" + col + ") > 0" : "") : " and datalength(" + col + ") > 0"));
                     PreparedStatement u = t.prepareStatement("update " + ts + "." + tb + " set " + col + " = ? where " + where)) {
                    while (r.next()) {
                        found++;
                        if (SKIP.contains(tb)) continue;
                        byte[] b = r.getBytes(match.length + 1); bytes += b.length;
                        u.setBytes(1, b);
                        for (int i = 0; i < match.length; i++) u.setObject(i + 2, r.getObject(i + 1));
                        if (u.executeUpdate() >= 1) upd++; else miss++;
                    }
                }
                t.commit();
                if (found > 0) {
                    String note = SKIP.contains(tb) ? "SKIPPED (redeploy)" : String.join("+", match);
                    w.println(tb + "," + col + "," + note + "," + found + "," + upd + "," + miss + "," + bytes + "," + c[2]);
                    System.out.println(tb + "." + col + ": " + found + " with bytes, " + upd + " copied, " + miss + " not found [" + note + "]");
                }
            }
        }
    }

    // Counts '' in the text columns of the listed tables on Db2, and on the source when it is SQL Server
    // (Oracle cannot store ''; SQL Server can, so there the two counts must match).
    static void empties(String out, String list) throws Exception {
        try (Connection s = src(); Connection t = tgt(); PrintWriter w = new PrintWriter(out, "UTF-8")) {
            String ts = env("TGT_SCHEMA"), ss = env("SRC_SCHEMA");
            boolean ora = s.getMetaData().getDatabaseProductName().toLowerCase().contains("oracle");
            w.println("table,column,source_rows,db2_rows");
            long total = 0, totalSrc = 0; int checked = 0, differ = 0;
            for (String tb : list.split(",")) {
                List<String> cs = new ArrayList<>();
                try (PreparedStatement p = t.prepareStatement(
                        "select colname from syscat.columns where tabschema = ? and tabname = ? and typename in ('VARCHAR','CHARACTER')")) {
                    p.setString(1, ts); p.setString(2, tb);
                    try (ResultSet r = p.executeQuery()) { while (r.next()) cs.add(r.getString(1)); }
                }
                for (String c : cs) {
                    long n = count(t, "select count(*) from " + ts + "." + tb + " where " + c + " = ''");
                    long m = ora ? 0 : count(s, "select count(*) from " + ss + "." + tb + " where " + c + " = ''");
                    checked++; total += Math.max(n, 0); totalSrc += Math.max(m, 0);
                    if (n != m) differ++;
                    if (n != 0 || m != 0) w.println(tb + "," + c + "," + m + "," + n);
                }
            }
            w.println("TOTAL," + checked + " columns," + totalSrc + "," + total);
            System.out.println(checked + " text columns checked; rows holding '': source " + totalSrc + ", Db2 " + total
                + "; columns that differ: " + differ + " -> " + out);
        }
    }

    // Replace the target rows of whole tables with the source rows (same ids, rowstamp from MAXSEQ).
    static void restore(String out, String list) throws Exception {
        try (Connection s = src(); Connection t = tgt(); PrintWriter w = new PrintWriter(out, "UTF-8")) {
            t.setAutoCommit(false);
            String ts = env("TGT_SCHEMA"), ss = env("SRC_SCHEMA");
            w.println("table,db2_rows_before,source_rows,inserted");
            for (String tb : list.split(",")) {
                long before = count(t, "select count(*) from " + ts + "." + tb);
                long srcRows = count(s, "select count(*) from " + ss + "." + tb);
                if (srcRows < 0 || before < 0) { w.println(tb + ",missing,missing,0"); continue; }
                int ins = 0;
                try (Statement st = t.createStatement()) { st.executeUpdate("delete from " + ts + "." + tb); }
                try (Statement q = s.createStatement(); ResultSet r = q.executeQuery("select * from " + ss + "." + tb)) {
                    ResultSetMetaData m = r.getMetaData();
                    StringBuilder names = new StringBuilder(), marks = new StringBuilder();
                    List<Integer> idx = new ArrayList<>();
                    for (int i = 1; i <= m.getColumnCount(); i++) {
                        String n = m.getColumnName(i).toUpperCase();
                        if (names.length() > 0) { names.append(','); marks.append(','); }
                        names.append(n);
                        if (n.equals("ROWSTAMP")) marks.append("NEXTVAL FOR " + ts + ".MAXSEQ");
                        else { marks.append('?'); idx.add(i); }
                    }
                    try (PreparedStatement p = t.prepareStatement("insert into " + ts + "." + tb + " (" + names + ") values (" + marks + ")")) {
                        while (r.next()) {
                            int k = 1;
                            for (int i : idx) {
                                Object v = r.getObject(i);
                                if (v instanceof Blob b) v = b.getBytes(1, (int) b.length());
                                else if (v instanceof Clob c) v = c.getSubString(1, (int) c.length());
                                p.setObject(k++, v);
                            }
                            ins += p.executeUpdate();
                        }
                    }
                }
                t.commit();
                w.println(tb + "," + before + "," + srcRows + "," + ins);
                System.out.println(tb + ": Db2 had " + before + ", source " + srcRows + ", restored " + ins);
            }
        }
    }
}

"""List the Maximo records IBM's update scripts write differently per database (DBC files with a target).

Created by Mohamed Chihaoui.

The maxdemo files (diff_ibm_platform_rows.py) only hold what the demo database contained when IBM unloaded
it. Records added later by update scripts are not in them. Those scripts are DBC files, and when a record
needs different SQL per database IBM ships two scripts: one with target="sqlserver" (or "oracle") and one
with target="not_sqlserver" (or "not_oracle"), inserting the same record with different text. Example in
Manage 9.3: V9300_44.dbc (sqlserver) inserts condition DLJOBITEMS with "select top 1 1 ...", V9300_45.dbc
(not_sqlserver) inserts it with "... fetch first 1 row only".

A third form lives in add-on content files (.isc): one text in "new" plus per-database substitution
rules, e.g. <ATTRIBUTE name="SELECTSTMT" new="select avg(sysdate-reportdate) from workorder"
db2value="{sysdate -> current timestamp}" sqsvalue="{sysdate -> getdate()}"/> (Nuclear KPIs).

This script pairs them by table and natural key and writes the same CSV as diff_ibm_platform_rows.py
(table,key,column,oracle_text,db2_text; the first text column holds the source platform's text), so
ibm_db2_text_fix.py turns it into guarded updates.

Usage: python diff_ibm_targeted_dbc.py <SMP>/maximo sqlserver|oracle <out.csv>
"""
import csv
import re
import sys
import xml.etree.ElementTree as ET
from pathlib import Path

from diff_ibm_platform_rows import KEYS

SQL_COLS = {"QUERY": "CLAUSE", "MAXRELATIONSHIP": "WHERECLAUSE", "CONDITION": "EXPRESSION", "ESCALATION": "CONDITION",
            "KPIMAIN": "CLAUSE", "MAXINTOBJECT": "RESTRICTWHERE", "OSLCQUERY": "CLAUSE", "MAXTABLEDOMAIN": "LISTWHERECLAUSE",
            "SCRIPTLAUNCHPOINT": "CONDITION", "SLA": "CONDITION", "WFCONDITION": "CONDITION"}

def target_of(path):
    head = path.read_text(encoding="utf-8", errors="replace")[:600]
    m = re.search(r'<script[^>]*\btarget="([a-z_,]+)"', head, re.S | re.I)
    return m.group(1).lower() if m else None

def records(path):
    """(table, key, column) -> text for every insertrow / update of a SQL-bearing column in one DBC file."""
    out = {}
    try:
        # IBM's DBC files carry Maximo binds such as :&USERNAME& unescaped, which is not valid XML
        text = re.sub(r"&(?!amp;|lt;|gt;|quot;|apos;|#)", "&amp;", path.read_text(encoding="utf-8", errors="replace"))
        root = ET.fromstring(text)
    except ET.ParseError:
        return out
    for ins in root.iter("insert"):
        tb = (ins.get("table") or "").upper()
        if tb not in SQL_COLS or tb not in KEYS:
            continue
        for row in ins.iter("insertrow"):
            vals = {cv.get("column", "").upper(): cv.get("string") for cv in row.iter("columnvalue")}
            col = SQL_COLS[tb]
            if vals.get(col) is None:
                continue
            key = "|".join(str(vals.get(k)) for k in KEYS[tb])
            out[(tb, key, col)] = vals[col]
    return out

PLATFORM_ATTR = {"sqlserver": "sqsvalue", "oracle": "oravalue", "db2": "db2value"}

def apply_rules(text, rules):
    """Apply '{a -> b}' substitution rules (one or more) as IBM's .isc files write them."""
    for a, b in re.findall(r"\{\s*(.+?)\s*->\s*(.*?)\s*\}", rules or ""):
        text = text.replace(a, b)
    return text

def isc_records(path, source):
    """(table, key, column) -> (source text, db2 text) for .isc attributes that carry per-database values."""
    out = {}
    try:
        root = ET.fromstring(re.sub(r"&(?!amp;|lt;|gt;|quot;|apos;|#)", "&amp;", path.read_text(encoding="utf-8", errors="replace")))
    except ET.ParseError:
        return out
    for tb_el in root:
        tb = (tb_el.get("tbname") or "").upper()
        if tb not in KEYS:
            continue
        for key_el in tb_el.iter("KEY"):
            attrs = {a.get("name", "").upper(): a for a in key_el.iter("ATTRIBUTE")}
            key = "|".join(str(key_el.get(k) or (attrs[k].get("new") if k in attrs else None)) for k in KEYS[tb])
            for col, a in attrs.items():
                if not any(a.get(v) for v in PLATFORM_ATTR.values()):
                    continue
                base = a.get("new") or ""
                src = apply_rules(base, a.get(PLATFORM_ATTR[source])) if source != "oracle" or a.get("oravalue") else base
                tgt = apply_rules(base, a.get("db2value"))
                if src != tgt:
                    out[(tb, key, col)] = (src, tgt)
    return out

def main(smp, source, out_csv):
    files = [p for p in Path(smp).rglob("*.dbc")]
    src, other = {}, {}
    for p in files:
        t = target_of(p)
        if not t:
            continue
        recs = records(p)
        if t == source:
            src.update(recs)
        elif t in (f"not_{source}", "db2"):
            other.update(recs)
    isc = {}
    for p in Path(smp).rglob("*.isc"):
        isc.update(isc_records(p, source))
    n = 0
    with open(out_csv, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["table", "key", "column", "oracle_text", "db2_text"])
        for k, text in sorted(src.items()):
            if k in other and other[k] != text:
                w.writerow([k[0], k[1], k[2], text, other[k]]); n += 1
        for k, (a, b) in sorted(isc.items()):
            w.writerow([k[0], k[1], k[2], a, b]); n += 1
    print(f"{len(files)} DBC files; {len(src)} records in target={source} scripts; {len(isc)} .isc values with per-database text; "
          f"{n} rows with a different Db2 text -> {out_csv}")

"""Write a DB2 script that puts IBM's DB2 wording back on IBM's own records after an Oracle-to-DB2 move.

Created by Mohamed Chihaoui.

Input : ibm_platform_diff.csv from diff_ibm_platform_rows.py (same Maximo version and add-ons
        as the converted database: the maxdemo files must come from the same build).
Output: ibm_db2_text_fix.sql, one guarded UPDATE per record. The guard (column = IBM's Oracle
        text) means a record the customer has changed is left alone and shows up in the
        finder report instead.

Usage: python ibm_db2_text_fix.py ibm_platform_diff.csv ibm_db2_text_fix.sql
"""
import csv
import re
import sys

# table -> (SQL-bearing columns, natural key columns in the order the diff wrote them)
TABLES = {
    "MAXRELATIONSHIP": (("WHERECLAUSE",), ("PARENT", "NAME")),
    "QUERY": (("CLAUSE",), ("APP", "CLAUSENAME", "OWNER")),
    "OSLCQUERY": (("CLAUSE",), ("INTOBJECTNAME", "CLAUSENAME")),
    "KPIMAIN": (("CLAUSE", "SELECTSTMT"), ("KPINAME",)),
    "CONDITION": (("EXPRESSION",), ("CONDITIONNUM",)),
    "ESCALATION": (("CONDITION",), ("ESCALATION",)),
    "MAXINTOBJECT": (("RESTRICTWHERE",), ("INTOBJECTNAME",)),
    "MAXTABLEDOMAIN": (("VALIDTNWHERECLAUSE", "LISTWHERECLAUSE"), ("DOMAINID", "OBJECTNAME", "ORGID", "SITEID")),
    "MAXPROP": (("MAXIMODEFAULT",), ("PROPNAME",)),
    "MAXPROPVALUE": (("PROPVALUE",), ("PROPNAME", "SERVERNAME", "SERVERHOST")),
}
WS = re.compile(r"\s+")

def lit(v):
    return "'" + v.replace("'", "''") + "'"

def main(src, out):
    n = 0
    lines = ["-- Generated by ibm_db2_text_fix.py (Created by Mohamed Chihaoui)",
             "-- Run on the converted DB2 database:  db2 -tvf ibm_db2_text_fix.sql", ""]
    for r in csv.DictReader(open(src, encoding="utf-8")):
        spec = TABLES.get(r["table"])
        if not spec or r["column"] not in spec[0]:
            continue
        if WS.sub(" ", r["oracle_text"]).strip().lower() == WS.sub(" ", r["db2_text"]).strip().lower():
            continue   # only spacing differs
        keys = r["key"].split("|")
        cond = []
        for c, v in zip(spec[1], keys):
            cond.append(f"{c} is null" if v == "None" else f"{c} = {lit(v)}")
        cond.append(f"{r['column']} = {lit(r['oracle_text'])}")
        lines.append(f"-- {r['table']} {r['key']}")
        lines.append(f"update {r['table'].lower()} set {r['column'].lower()} = {lit(r['db2_text'])}\n  where "
                     + "\n    and ".join(cond) + ";")
        n += 1
    lines += ["", "commit;"]
    open(out, "w", encoding="utf-8", newline="\n").write("\n".join(lines) + "\n")
    print(n, "guarded updates ->", out)

"""Dual-run check: run every saved query on the Oracle or SQL Server source and on the Db2 target, compare row counts.

Created by Mohamed Chihaoui.

A rewritten clause is only "converted" when Db2 returns what Oracle returned. This script reads the
saved queries (QUERY) from the source, turns each into
    select count(*) from <application main table> where <clause>
and runs it on both databases. On Db2 it uses the clause stored in the Db2 database, or the rewrite
from --rewrites (CSV: app,clausename,owner,db2_clause,why) when you are testing a draft.

It drives the two command-line clients, so it runs on any host that has them:
  --src         oracle (default) or sqlserver
  --ora-shell   a command that runs a bash script where sqlplus (or sqlcmd) works  (default: bash)
  --db2-shell   a command that runs a bash script where the db2 CLP works  (default: bash)
  --ora-conn    sqlplus connect string, e.g. maximo/<password>@//<host>:1521/<service>   (required)
  --ss-conn     sqlcmd options for a SQL Server source, e.g. "-S <host> -U maximo -P <password> -d MAXDB"  (with --src sqlserver)
  --db2-db / --db2-schema                          (default: MAXDB / MAXIMO)
Example, from a jump host:
  python dual_run.py --ora-shell "ssh oracle-host bash -s" --db2-shell "ssh db2inst1@db2-host bash -s" \\
                     --ora-conn "maximo/<password>@//<host>:1521/<service>" --rewrites db2_rewrites.csv --out dual_run.csv

Bind variables: :USER becomes 'MAXADMIN'; clauses with other Maximo binds (:&...&, :siteid ...) are
reported as SKIPPED, test those in the application.
"""
import argparse
import csv
import re
import shlex
import subprocess

SEP = "~"
A = None   # parsed arguments

def sh(shell, script):
    """Run a bash script through a shell command (local bash, ssh, ...).
    The script is written to a temp file before it runs: the db2 command line processor reads
    stdin and would swallow the rest of a script piped into bash. Carriage returns are removed."""
    wrapped = ("f=$(mktemp); cat > \"$f\" <<'__DUALRUN__'\n" + script.replace("\r", "")
               + "\n__DUALRUN__\nbash \"$f\"; rm -f \"$f\"\n")
    # bytes, not text: on Windows a text-mode pipe turns every \n into \r\n
    out = subprocess.run(shlex.split(shell), input=wrapped.encode("utf-8"), capture_output=True).stdout
    return out.decode("utf-8", errors="replace").replace("\r", "")

def sqlplus(body):
    return ("sqlplus -s " + A.ora_conn + " <<'EOF'\nset pagesize 0 linesize 32767 trimspool on feedback off heading off\n"
            + body + "\nexit\nEOF\n")

def db2_head():
    return [". ~/sqllib/db2profile 2>/dev/null", f"db2 connect to {A.db2_db} > /dev/null", f"db2 set schema {A.db2_schema} > /dev/null"]

def sqlcmd(body, sep=False):
    """sqlcmd script for a SQL Server source. Each statement is its own batch (go), so one error does not stop the rest."""
    opts = '-h -1 -W -s "~"' if sep else "-h -1 -W"
    return f"{A.sqlcmd} -C {A.ss_conn} {opts} -i /dev/stdin <<'EOF'\nset nocount on\ngo\n{body}\nEOF\n"

def queries():
    if A.src == "sqlserver":
        sql = ("select q.app, q.clausename, q.owner, a.maintbname, replace(replace(q.clause, char(10), ' '), char(13), ' ') "
               "from query q join maxapps a on a.app = q.app where a.maintbname is not null order by 1, 2, 3\ngo")
        return [l.split(SEP) for l in sh(A.ora_shell, sqlcmd(sql, sep=True)).splitlines() if l.count(SEP) >= 4]
    sql = ("set markup csv on quote off delimiter ~\n"
           "select q.app, q.clausename, q.owner, a.maintbname, replace(replace(q.clause, chr(10), ' '), chr(13), ' ') "
           "from query q join maxapps a on a.app = q.app where a.maintbname is not null order by 1, 2, 3;")
    return [l.split(SEP) for l in sh(A.ora_shell, sqlplus(sql)).splitlines() if SEP in l]

def db2_clauses():
    lines = db2_head() + ["db2 -x \"select app || '~' || clausename || '~' || owner || '~' || "
                          "replace(replace(clause, chr(10), ' '), chr(13), ' ') from query\""]
    d = {}
    for l in sh(A.db2_shell, "\n".join(lines)).splitlines():
        p = l.rstrip().split(SEP, 3)
        if len(p) == 4:
            d[tuple(p[:3])] = p[3]
    return d

def prep(clause):
    # Maximo accepts a stored clause that starts with WHERE or ends with ORDER BY; count(*) does not
    clause = re.sub(r"^\s*where\s+", "", clause, flags=re.I)
    clause = re.sub(r"\s+order\s+by\s+[\w\s,.]+$", "", clause, flags=re.I)
    if re.search(r":&|:[a-z]", re.sub(r":user\b", "", clause, flags=re.I), re.I):
        return None
    return re.sub(r":user\b", "'MAXADMIN'", clause, flags=re.I)

def count_oracle(items):
    if A.src == "sqlserver":
        lines = []
        for i, (tbl, clause) in enumerate(items):
            # the marker gets its own batch: a syntax error stops the whole batch, print included
            lines += [f"print '@@{i}'", "go", f"select count(*) from {tbl} where {clause}", "go"]
        return parse(sh(A.ora_shell, sqlcmd("\n".join(lines))), len(items), r"(Msg \d+)")
    lines = []
    for i, (tbl, clause) in enumerate(items):
        lines += [f"prompt @@{i}", f"select count(*) from {tbl} where {clause};"]
    return parse(sh(A.ora_shell, sqlplus("\n".join(lines))), len(items), r"(ORA-\d+)")

def count_db2(items):
    lines = db2_head()
    for i, (tbl, clause) in enumerate(items):
        stmt = f"select count(*) from {tbl} where {clause}".replace('"', '\\"')
        lines.append(f'echo "@@{i}"; db2 -x "{stmt}" 2>&1')
    return parse(sh(A.db2_shell, "\n".join(lines)), len(items), r"(SQL\d+N)")

def parse(out, n, err):
    res, cur, buf = {}, None, []
    for line in out.splitlines() + ["@@end"]:
        m = re.match(r"^@@(\d+|end)", line.strip())
        if m:
            if cur is not None:
                e = re.search(err, " ".join(buf))
                num = re.search(r"^\s*(\d+)\s*$", "\n".join(buf), re.M)
                res[cur] = e.group(1) if e else (num.group(1) if num else "?")
            cur, buf = (None if m.group(1) == "end" else int(m.group(1))), []
        elif cur is not None:
            buf.append(line)
    return [res.get(i, "?") for i in range(n)]

def main():
    global A
    ap = argparse.ArgumentParser()
    ap.add_argument("--ora-shell", default="bash")
    ap.add_argument("--db2-shell", default="bash")
    ap.add_argument("--src", choices=["oracle", "sqlserver"], default="oracle")
    ap.add_argument("--ora-conn", help="sqlplus connect string (Oracle source)")
    ap.add_argument("--ss-conn", help='sqlcmd connection options (SQL Server source), e.g. "-S host -U maximo -P pw -d MAXDB"')
    ap.add_argument("--sqlcmd", default="sqlcmd", help="path of sqlcmd on the source shell")
    ap.add_argument("--db2-db", default="MAXDB")
    ap.add_argument("--db2-schema", default="MAXIMO")
    ap.add_argument("--rewrites")
    ap.add_argument("--only", help="regexp on clausename")
    ap.add_argument("--out", default="dual_run.csv")
    A = ap.parse_args()
    if (A.src == "oracle" and not A.ora_conn) or (A.src == "sqlserver" and not A.ss_conn):
        ap.error("--ora-conn is required for an Oracle source, --ss-conn for a SQL Server source")
    rew = {}
    if A.rewrites:
        for r in csv.DictReader(open(A.rewrites, encoding="utf-8")):
            rew[(r["app"], r["clausename"], r["owner"])] = r["db2_clause"]
    on_db2 = db2_clauses()
    rows, o_items, d_items = [], [], []
    for app, name, owner, tbl, clause in queries():
        if A.only and not re.search(A.only, name):
            continue
        key = (app, name, owner)
        d_clause = rew.get(key, on_db2.get(key, clause))
        po, pd = prep(clause), prep(d_clause)
        rows.append([app, name, owner, tbl, clause, d_clause, po is None])
        if po is not None:
            o_items.append((tbl, po)); d_items.append((tbl, pd))
    oc, dc = count_oracle(o_items), count_db2(d_items)
    k, tally = 0, {}
    with open(A.out, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(["app", "clausename", "owner", "table", "oracle_count", "db2_count", "verdict", "oracle_clause", "db2_clause"])
        for app, name, owner, tbl, clause, d_clause, skipped in rows:
            if skipped:
                o = d = ""; v = "SKIPPED (binds)"
            else:
                o, d = oc[k], dc[k]; k += 1
                v = ("DB2 ERROR" if d.startswith("SQL") else "SOURCE ERROR" if o.startswith(("ORA", "Msg")) else
                     "SAME" if o == d else "DIFFERENT")
            tally[v] = tally.get(v, 0) + 1
            w.writerow([app, name, owner, tbl, o, d, v, clause, d_clause])
    print(tally, "->", A.out)

if __name__ == "__main__":
    main()
