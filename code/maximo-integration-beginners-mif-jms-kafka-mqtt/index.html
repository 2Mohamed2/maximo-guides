# produce.py
import json
from kafka import KafkaProducer
from kc import cfg
p = KafkaProducer(**cfg(), key_serializer=str.encode,
                  value_serializer=lambda v: json.dumps(v).encode(), acks="all")
for i, pump in enumerate(["PUMP-001", "PUMP-002", "PUMP-001", "PUMP-003", "PUMP-001"]):
    md = p.send("demo.pump.alerts", key=pump,
                value={"pump": pump, "alert": "HIGH_VIBRATION", "seq": i}).get(15)
    print(f"sent key={pump} seq={i} -> partition {md.partition} offset {md.offset}")
p.flush()

# consume.py
import sys
from kafka import KafkaConsumer
from kc import cfg
group = sys.argv[1]
c = KafkaConsumer("demo.pump.alerts", **cfg(), group_id=group,
                  auto_offset_reset="earliest", consumer_timeout_ms=12000)
n = 0
for m in c:
    n += 1
    print(f"[{group}] partition {m.partition} offset {m.offset} key={m.key.decode()} value={m.value.decode()}")
print(f"[{group}] read {n} messages")
c.close()

# pump.py
import json, random, ssl, time
import paho.mqtt.client as mqtt
HOST = "wsp.messaging.iot.<your-suite-domain>"
c = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2, client_id="d:wsp:CoolingPump:PUMP-001")
c.username_pw_set("use-token-auth", open("PUMP-001.token").read().strip())
ctx = ssl.create_default_context()
ctx.check_hostname = False; ctx.verify_mode = ssl.CERT_NONE   # lab only: load the suite CA instead
c.tls_set_context(ctx)
c.on_connect = lambda cl, u, flags, rc, props=None: print("connected:", rc)
c.connect(HOST, 443, keepalive=60)
c.loop_start(); time.sleep(2)
while True:
    reading = {"temperature": round(62 + random.uniform(-1, 1), 2),
               "pressure":    round(4.1 + random.uniform(-.1, .1), 3),
               "vibration":   round(2.1 + random.uniform(-.25, .25), 3),
               "flow":        round(120 + random.uniform(-2, 2), 2)}
    c.publish("iot-2/evt/status/fmt/json", json.dumps(reading), qos=1).wait_for_publish(10)
    print(time.strftime("%H:%M:%S"), reading)
    time.sleep(10)

// One pump reading. While flow.fault is true the bearing wears a little more on every reading.
const fault = flow.get('fault') || false;
let wear = fault ? (flow.get('wear') || 0) + 0.25 : 0;
flow.set('wear', wear);
const t = Date.now() / 1000;
const jitter = (n) => (Math.random() - 0.5) * n;
msg.payload = {
    temperature: +(61 + 3 * Math.sin(t / 100) + jitter(1.2) + wear * 2.5).toFixed(2),
    pressure:    +(4.2 + 0.15 * Math.sin(t / 70) + jitter(0.1) - wear * 0.08).toFixed(3),
    vibration:   +(2.1 + jitter(0.5) + wear).toFixed(3),
    flow:        +(118 + 4 * Math.sin(t / 125) + jitter(2) - wear * 3).toFixed(2)
};
msg.topic = 'iot-2/evt/status/fmt/json';
node.status({ fill: fault ? 'red' : 'green', shape: 'dot', text: 'vibration ' + msg.payload.vibration });
return msg;

// A legacy system posts one line of text: temperature,pressure,vibration,flow
const p = String(msg.payload).trim().split(',').map(Number);
if (p.length !== 4 || p.some(isNaN)) {
    msg.statusCode = 400;
    msg.payload = { error: 'expected: temperature,pressure,vibration,flow' };
    return [null, msg];
}
const reading = { temperature: p[0], pressure: p[1], vibration: p[2], flow: p[3] };
const toIot = { topic: 'iot-2/evt/status/fmt/json', payload: reading };
msg.statusCode = 202;
msg.payload = { accepted: true, device: 'PUMP-001', reading: reading };
return [toIot, msg];
